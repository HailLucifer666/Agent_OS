```tsx
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    let animationId: number;
    let time = 0;

    const resize = () => {
      canvas.width = canvas.offsetWidth * window.devicePixelRatio;
      canvas.height = canvas.offsetHeight * window.devicePixelRatio;
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
    };

    resize();
    window.addEventListener("resize", resize);

    const animate = () => {
      if (!isPaused) {
        time += 0.005;
      }

      const width = canvas.offsetWidth;
      const height = canvas.offsetHeight;

      ctx.clearRect(0, 0, width, height);

      // Draw subtle background gradient
      const gradient = ctx.createRadialGradient(width / 2, height / 2, 0, width / 2, height / 2, width);
      gradient.addColorStop(0, "rgba(30, 20, 60, 0.3)");
      gradient.addColorStop(1, "rgba(15, 15, 18, 0)");
      ctx.fillStyle = gradient;
      ctx.fillRect(0, 0, width, height);

      // Draw nodes
      nodes.forEach((node) => {
        const x = width / 2 + node.x * Math.cos(time + node.id.charCodeAt(6) * 0.1) * 0.3;
        const y = height / 2 + node.y * Math.sin(time + node.id.charCodeAt(6) * 0.1) * 0.3;

        const pulseScale = node.pulse ? 1 + Math.sin(time * 3) * 0.2 : 1;

        ctx.beginPath();
        ctx.arc(x, y, node.size * pulseScale, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(200, 220, 255, ${node.opacity})`;
        ctx.fill();

        // Glow for larger nodes
        if (node.size > 5) {
          ctx.beginPath();
          ctx.arc(x, y, node.size * 3 * pulseScale, 0, Math.PI * 2);
          const glow = ctx.createRadialGradient(x, y, 0, x, y, node.size * 3);
          glow.addColorStop(0, `rgba(150, 200, 255, ${node.opacity * 0.3})`);
          glow.addColorStop(1, "rgba(150, 200, 255, 0)");
          ctx.fillStyle = glow;
          ctx.fill();
        }
      });
```

### `components/goals/Goals.tsx`

```tsx
"use client";

import { useState } from "react";
import { Target, Mic, Plus, Sparkles } from "lucide-react";
import { cn } from "@/lib/utils";

interface Goal {
  id: string;
  title: string;
  category: string | null;
  status: "active" | "done";
}

const initialGoals: Goal[] = [
  { id: "1", title: "Launch Agent OS v1", category: "Product", status: "done" },
  { id: "2", title: "Integrate Claude 3.5 Sonnet", category: "Integration", status: "done" },
];

export default function Goals() {
  const [goals, setGoals] = useState<Goal[]>(initialGoals);
  const [newGoal, setNewGoal] = useState("");
  const [filter, setFilter] = useState<"active" | "done" | "all">("all");

  const stats = {
    total: goals.length,
    active: goals.filter((g) => g.status === "active").length,
    completed: goals.filter((g) => g.status === "done").length,
    progress: goals.length > 0 ? Math.round((goals.filter((g) => g.status === "done").length / goals.length) * 100) : 0,
  };

  const filteredGoals = goals.filter((g) => {
    if (filter === "active") return g.status === "active";
    if (filter === "done") return g.status === "done";
    return true;
  });

  const addGoal = () => {
    if (!newGoal.trim()) return;
    setGoals([...goals, { id: Date.now().toString(), title: newGoal, category: null, status: "active" }]);
    setNewGoal("");
  };

  const toggleGoal = (id: string) => {
    setGoals(goals.map((g) => (g.id === id ? { ...g, status: g.status === "active" ? "done" : "active" } : g)));
  };

  return (
    <div className="p-8 max-w-[1400px]">
      {/* Header */}
      <div className="mb-8">
        <div className="flex items-center justify-between mb-4">
          <div className="flex items-center gap-3">
            <span className="roman-numeral">X.</span>
            <span className="text-[0.6875rem] uppercase tracking-[0.2em] text-text-tertiary">Self · Goals</span>
          </div>
          <div className="flex items-center gap-3">
            <button className="flex items-center gap-2 px-3 py-1.5 rounded-md bg-background-secondary border border-border-color text-xs text-text-secondary hover:border-border-hover transition-colors">
              <span>⌘K</span>
              <span>Command palette</span>
            </button>
            <button className="flex items-center gap-2 px-3 py-1.5 rounded-md bg-background-secondary border border-border-color text-xs text-text-secondary hover:border-border-hover transition-colors">
              <Sparkles className="w-3 h-3" />
              <span>ALL SYSTEMS</span>
            </button>
          </div>
        </div>

        <h1 className="text-5xl font-bold text-text-primary mb-3 tracking-tight">Goals</h1>
        <p className="text-lg text-text-secondary mb-2">Set targets. Tick them off. Watch the bar fill. Saved to Goals.md.</p>
        <p className="text-xs text-accent-gold font-mono">07:78 · LOCAL · STUDIO</p>
      </div>

      {/* Stats */}
      <div className="grid grid-cols-4 gap-4 mb-8">
        <div className="bg-gradient-to-br from-background-secondary to-background-tertiary/50 border border-border-color rounded-xl p-5">
          <div className="text-[0.625rem] uppercase tracking-wider text-text-tertiary mb-2">Total</div>
          <div className="text-3xl font-bold text-text-primary">{stats.total}</div>
        </div>
        <div className="bg-gradient-to-br from-background-secondary to-background-tertiary/50 border border-border-color rounded-xl p-5">
          <div className="text-[0.625rem] uppercase tracking-wider text-text-tertiary mb-2">Active</div>
          <div className="text-3xl font-bold text-accent-purple">{stats.active}</div>
        </div>
        <div className="bg-gradient-to-br from-background-secondary to-background-tertiary/50 border border-border-color rounded-xl p-5">
          <div className="text-[0.625rem] uppercase tracking-wider text-text-tertiary mb-2">Completed</div>
          <div className="text-3xl font-bold text-accent-teal">{stats.completed}</div>
        </div>
        <div className="bg-gradient-to-br from-background-secondary to-background-tertiary/50 border border-border-color rounded-xl p-5">
          <div className="text-[0.625rem] uppercase tracking-wider text-text-tertiary mb-2">Progress</div>
          <div className="text-3xl font-bold text-accent-gold">{stats.progress}%</div>
          <div className="w-full h-1.5 bg-background-primary rounded-full mt-3 overflow-hidden">
            <div
              className="h-full bg-accent-gold rounded-full transition-all duration-500"
              style={{ width: `${stats.progress}%` }}
            />
          </div>
        </div>
      </div>

      {/* Add Goal */}
      <div className="bg-background-secondary border border-border-color rounded-xl p-5 mb-6">
        <div className="flex items-center gap-2 mb-3">
          <Target className="w-4 h-4 text-accent-teal" />
          <span className="text-sm font-medium text-text-primary">Add a goal</span>
        </div>
        <div className="flex items-center gap-3">
          <button className="w-9 h-9 rounded-lg bg-background-tertiary flex items-center justify-center text-text-tertiary hover:text-text-primary transition-colors">
            <Mic className="w-4 h-4" />
          </button>
          <input
            type="text"
            value={newGoal}
            onChange={(e) => setNewGoal(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && addGoal()}
            placeholder="What's the goal? (Enter to add)"
            className="flex-1 bg-background-primary rounded-lg border border-border-color px-4 py-2.5 text-sm text-text-primary placeholder:text-text-muted outline-none focus:border-accent-gold/50 transition-colors"
          />
          <select className="bg-background-primary rounded-lg border border-border-color px-3 py-2.5 text-sm text-text-secondary outline-none">
            <option>No category</option>
            <option>Product</option>
            <option>Integration</option>
            <option>Research</option>
          </select>
          <button
            onClick={addGoal}
            className="flex items-center gap-2 px-4 py-2.5 rounded-lg bg-accent-teal/10 text-accent-teal text-sm border border-accent-teal/20 hover:bg-accent-teal/20 transition-colors"
          >
            <Plus className="w-4 h-4" />
            <span>Add</span>
          </button>
        </div>
      </div>

      {/* Filters */}
      <div className="flex items-center justify-between mb-6">
        <div className="flex items-center gap-2">
          {(["active", "done", "all"] as const).map((f) => (
            <button
              key={f}
              onClick={() => setFilter(f)}
              className={cn(
                "px-3 py-1.5 rounded-full text-xs transition-colors capitalize",
                filter === f
                  ? "bg-accent-purple/10 text-accent-purple border border-accent-purple/20"
                  : "text-text-tertiary hover:text-text-secondary"
              )}
            >
              {f} · {f === "active" ? stats.active : f === "done" ? stats.completed : stats.total}
            </button>
          ))}
        </div>
        <span className="text-[0.625rem] uppercase tracking-wider text-text-muted">
          Saved to Agentic OS/Goals.md
        </span>
      </div>

      {/* Goals List */}
      {filteredGoals.length === 0 ? (
        <div className="flex items-center justify-center py-20">
          <p className="text-sm text-text-secondary">Nothing to chase right now. Add one above.</p>
        </div>
      ) : (
        <div className="space-y-2">
          {filteredGoals.map((goal) => (
            <div
              key={goal.id}
              onClick={() => toggleGoal(goal.id)}
              className="flex items-center gap-3 p-4 rounded-lg bg-background-secondary border border-border-color hover:border-border-hover cursor-pointer transition-colors"
            >
              <div
                className={cn(
                  "w-5 h-5 rounded border-2 flex items-center justify-center transition-colors",
                  goal.status === "done"
                    ? "bg-accent-teal border-accent-teal"
                    : "border-text-muted"
                )}
              >
                {goal.status === "done" && <span className="text-white text-xs">✓</span>}
              </div>
              <span
                className={cn(
                  "text-sm transition-colors",
                  goal.status === "done" ? "text-text-muted line-through" : "text-text-primary"
                )}
              >
                {goal.title}
              </span>
              {goal.category && (
                <span className="ml-auto px-2 py-0.5 rounded-full bg-background-tertiary text-xs text-text-tertiary">
                  {goal.category}
                </span>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```
